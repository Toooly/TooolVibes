```tsx
"use client";

import { Canvas, useFrame } from "@react-three/fiber";
import {
  Float,
  MeshDistortMaterial,
  TorusKnot,
} from "@react-three/drei";
import { EffectComposer, Bloom } from "@react-three/postprocessing";
import { useRef } from "react";
import * as THREE from "three";

function Ribbon({
  color,
  radius,
  speed,
  offset,
}: {
  color: string;
  radius: number;
  speed: number;
  offset: number;
}) {
  const ref = useRef<THREE.Mesh>(null);

  useFrame(({ clock }) => {
    if (!ref.current) return;
    const t = clock.getElapsedTime() * speed + offset;
    ref.current.rotation.x = Math.sin(t * 0.4) * 0.3;
    ref.current.rotation.y = t;
    ref.current.rotation.z = Math.cos(t * 0.3) * 0.2;
  });

  return (
    <mesh ref={ref}>
      <torusGeometry args={[radius, 0.04, 12, 80]} />
      <meshStandardMaterial
        color={color}
        emissive={color}
        emissiveIntensity={0.6}
        roughness={0.2}
        metalness={0.8}
        transparent
        opacity={0.9}
      />
    </mesh>
  );
}

function LogoScene() {
  const knotRef = useRef<THREE.Mesh>(null);

  useFrame(({ clock }) => {
    if (!knotRef.current) return;
    knotRef.current.rotation.x = clock.getElapsedTime() * 0.15;
    knotRef.current.rotation.y = clock.getElapsedTime() * 0.25;
  });

  return (
    <>
      <ambientLight intensity={0.4} />
      <pointLight position={[4, 4, 4]} color="#6a3cff" intensity={80} />
      <pointLight position={[-4, -2, 4]} color="#39d2f6" intensity={60} />
      <pointLight position={[0, -4, -2]} color="#f3bf4c" intensity={40} />

      <Float speed={1.5} rotationIntensity={0.4} floatIntensity={0.6}>
        <TorusKnot
          ref={knotRef}
          args={[1.2, 0.35, 128, 32]}
          scale={1.6}
        >
          <MeshDistortMaterial
            color="#6a3cff"
            emissive="#39d2f6"
            emissiveIntensity={0.25}
            roughness={0.15}
            metalness={0.7}
            distort={0.25}
            speed={1.2}
          />
        </TorusKnot>
      </Float>

      <Ribbon color="#39d2f6" radius={2.0} speed={0.6} offset={0} />
      <Ribbon color="#f3bf4c" radius={2.3} speed={-0.5} offset={Math.PI} />

      <EffectComposer>
        <Bloom
          intensity={0.4}
          threshold={0.85}
          luminanceSmoothing={0.9}
          mipmapBlur
        />
      </EffectComposer>
    </>
  );
}

export function Logo3DIntro() {
  return (
    <div className="brand-intro__logo-stage">
      <Canvas
        dpr={[1, 1.5]}
        gl={{ antialias: false, alpha: true }}
        camera={{ position: [0, 0, 6], fov: 45 }}
        style={{
          width: "100%",
          height: "100%",
        }}
      >
        <LogoScene />
      </Canvas>
    </div>
  );
}
```

```tsx
"use client";

import { usePathname } from "next/navigation";
import { useState, useEffect, useCallback, useRef } from "react";
import { motion, AnimatePresence } from "motion/react";
import { X } from "lucide-react";
import { Logo3DIntro } from "@/components/3d/logo-3d-intro";
import { SOUND_PLAY_EVENT } from "@/lib/gamification/sound-engine";

const INTRO_DURATION_MS = 3800;
const HAS_SEEN_INTRO_KEY = "tooolvibes_has_seen_intro";

type IntroPhase = "entering" | "playing" | "leaving" | "unmounted";

function dispatchIntroFinished(skipped: boolean) {
  if (typeof window === "undefined") return;
  window.dispatchEvent(
    new CustomEvent("tooolvibes:brand-intro-finished", {
      detail: { skipped },
    })
  );
}

function dispatchSound(sound: string) {
  if (typeof window === "undefined") return;
  window.dispatchEvent(
    new CustomEvent(SOUND_PLAY_EVENT, {
      detail: { sound },
    })
  );
}

function StarField() {
  const stars = useRef(
    Array.from({ length: 40 }, (_, i) => ({
      id: i,
      left: `${Math.random() * 100}%`,
      top: `${Math.random() * 100}%`,
      duration: `${2 + Math.random() * 4}s`,
      delay: `${Math.random() * 3}s`,
    }))
  ).current;

  return (
    <div className="brand-intro__stars" aria-hidden="true">
      {stars.map((star) => (
        <span
          key={star.id}
          className="brand-intro__star"
          style={{
            left: star.left,
            top: star.top,
            animationDuration: star.duration,
            animationDelay: star.delay,
          }}
        />
      ))}
    </div>
  );
}

export function HomeBrandIntro({
  onComplete,
}: {
  onComplete?: (skipped: boolean) => void;
}) {
  const pathname = usePathname();
  const [phase, setPhase] = useState<IntroPhase>("entering");
  const [shouldRender, setShouldRender] = useState(false);
  const hasFinished = useRef(false);

  const finishIntro = useCallback(
    (skipped = false) => {
      if (hasFinished.current) return;
      hasFinished.current = true;
      setPhase("leaving");

      setTimeout(() => {
        setPhase("unmounted");
        dispatchIntroFinished(skipped);
        onComplete?.(skipped);
      }, 700);
    },
    [onComplete]
  );

  useEffect(() => {
    if (typeof window === "undefined") return;

    const reducedMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)"
    ).matches;
    const hasSeen = sessionStorage.getItem(HAS_SEEN_INTRO_KEY);

    if (reducedMotion || hasSeen === "1") {
      setShouldRender(false);
      dispatchIntroFinished(true);
      onComplete?.(true);
      return;
    }

    setShouldRender(true);
    sessionStorage.setItem(HAS_SEEN_INTRO_KEY, "1");

    const timers: number[] = [];
    timers.push(
      window.setTimeout(() => dispatchSound("spark"), 200),
      window.setTimeout(() => dispatchSound("reveal"), 500),
      window.setTimeout(() => setPhase("playing"), 400),
      window.setTimeout(() => dispatchSound("finish"), INTRO_DURATION_MS - 400),
      window.setTimeout(() => finishIntro(false), INTRO_DURATION_MS)
    );

    return () => {
      timers.forEach((t) => window.clearTimeout(t));
    };
  }, [finishIntro, onComplete, pathname]);

  if (!shouldRender || phase === "unmounted") return null;

  const title = "TooolVibes";

  return (
    <div
      className={`brand-intro brand-intro--${phase}`}
      role="dialog"
      aria-label="Introduzione TooolVibes"
      aria-hidden={phase === "leaving"}
    >
      <StarField />

      <button
        className="brand-intro__skip"
        onClick={() => finishIntro(true)}
        aria-label="Salta introduzione"
        type="button"
      >
        <X size={20} />
```

```tsx
function CountdownTimer({
  duration,
  isPlaying,
  onComplete,
}: {
  duration: number;
  isPlaying: boolean;
  onComplete: () => void;
}) {
  const [progress, setProgress] = useState(1);
  const startRef = useRef<number | null>(null);
  const rafRef = useRef<number | null>(null);

  useEffect(() => {
    if (!isPlaying) return;
    startRef.current = performance.now();

    const tick = (now: number) => {
      const elapsed = now - (startRef.current ?? now);
      const remaining = Math.max(0, duration - elapsed);
      const p = remaining / duration;
      setProgress(p);

      if (remaining <= 0) {
        onComplete();
        return;
      }
      rafRef.current = requestAnimationFrame(tick);
    };

    rafRef.current = requestAnimationFrame(tick);
    return () => {
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
    };
  }, [isPlaying, duration, onComplete]);

  const radius = 24;
  const circumference = 2 * Math.PI * radius;
  const offset = circumference * (1 - progress);

  return (
    <svg width="56" height="56" viewBox="0 0 56 56" className="radar-console__timer-circle">
      <defs>
        <linearGradient id="radar-timer-gradient" x1="0" y1="0" x2="1" y2="1">
```

```tsx
function SignalSparkline({ values }: { values: number[] }) {
  const width = 120;
  const height = 40;
  const max = Math.max(...values, 1);
  const min = Math.min(...values, 0);
  const range = max - min || 1;

  const points = values
    .map((v, i) => {
      const x = (i / (values.length - 1 || 1)) * width;
      const y = height - ((v - min) / range) * height;
```

```tsx
import { useEffect, useRef, useState } from "react";

function useMouseParallax() {
  const [pos, setPos] = useState({ x: 0, y: 0 });
  const rafRef = useRef<number | null>(null);
  const pendingRef = useRef({ x: 0, y: 0 });

  useEffect(() => {
    const handle = (e: MouseEvent) => {
      const x = (e.clientX / window.innerWidth - 0.5) * 2;
      const y = (e.clientY / window.innerHeight - 0.5) * 2;
      pendingRef.current = { x, y };

      if (rafRef.current == null) {
        rafRef.current = requestAnimationFrame(() => {
          setPos(pendingRef.current);
          rafRef.current = null;
        });
      }
    };

    window.addEventListener("mousemove", handle);
    return () => window.removeEventListener("mousemove", handle);
  }, []);

  return pos;
}
```

Mantieni la stessa interfaccia pubblica (esportazioni, `SOUND_PLAY_EVENT`) in modo da non rompere i consumer.

Esempio di implementazione:

```ts
"use client";

import { useEffect, useRef, useCallback } from "react";

export const SOUND_PLAY_EVENT = "tooolvibes:play-sound";

type SoundName = "click" | "hover" | "transition" | "scan" | "reveal" | "spark" | "finish" | "level-up" | "badge";

const MUTED_KEY = "tooolvibes_sound_muted";

let audioCtx: AudioContext | null = null;
let muted = false;

function getAudioContext(): AudioContext {
  if (typeof window === "undefined") throw new Error("AudioContext only in browser");
  if (!audioCtx) {
    audioCtx = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)();
  }
  if (audioCtx.state === "suspended") {
    audioCtx.resume();
  }
  return audioCtx;
}

function beep(
  frequency: number,
  type: OscillatorType,
  duration: number,
  startGain: number,
  endGain: number
) {
  if (muted) return;
  const ctx = getAudioContext();
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();

  osc.type = type;
  osc.frequency.setValueAtTime(frequency, ctx.currentTime);

  gain.gain.setValueAtTime(startGain, ctx.currentTime);
  gain.gain.exponentialRampToValueAtTime(endGain, ctx.currentTime + duration);

  osc.connect(gain);
  gain.connect(ctx.destination);

  osc.start(ctx.currentTime);
  osc.stop(ctx.currentTime + duration);
}

export function playGameSound(name: SoundName) {
  if (muted) return;
  switch (name) {
    case "click":
      beep(800, "sine", 0.08, 0.0001, 0.15);
      break;
    case "hover":
      beep(600, "sine", 0.05, 0.0001, 0.08);
      break;
    case "transition":
      beep(400, "sawtooth", 0.3, 0.0001, 0.1);
      break;
    case "scan":
      beep(900, "sine", 0.4, 0.0001, 0.12);
      break;
    case "reveal":
      beep(1200, "triangle", 0.25, 0.0001, 0.15);
      break;
    case "spark":
      beep(1500, "sine", 0.1, 0.0001, 0.1);
      break;
    case "finish":
      beep(500, "square", 0.15, 0.0001, 0.18);
      setTimeout(() => beep(800, "square", 0.25, 0.0001, 0.12), 120);
      break;
    case "level-up":
      beep(523, "triangle", 0.15, 0.0001, 0.2);
      setTimeout(() => beep(659, "triangle", 0.15, 0.0001, 0.2), 150);
      setTimeout(() => beep(784, "triangle", 0.3, 0.0001, 0.15), 300);
      break;
    case "badge":
      beep(1000, "sine", 0.12, 0.0001, 0.15);
      break;
  }
}

export function isSoundMuted(): boolean {
  if (typeof window === "undefined") return false;
  return window.localStorage.getItem(MUTED_KEY) === "1";
}

export function setSoundMuted(value: boolean) {
  if (typeof window === "undefined") return;
  muted = value;
  window.localStorage.setItem(MUTED_KEY, value ? "1" : "0");
}

export function useGameSound() {
  useEffect(() => {
    muted = isSoundMuted();

    const handler = (e: Event) => {
      const detail = (e as CustomEvent).detail as { sound?: SoundName } | undefined;
      if (detail?.sound) {
        playGameSound(detail.sound);
      }
    };

    window.addEventListener(SOUND_PLAY_EVENT, handler);
    return () => window.removeEventListener(SOUND_PLAY_EVENT, handler);
  }, []);
}
```
